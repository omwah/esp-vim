# Vim as an ESP-IDF component.
#
# The source list mirrors BASIC_SRC in Vim's own src/Makefile. Every file is
# compiled at every feature level upstream -- the ones that do not apply here
# (GUI, X11, pango, cairo) reduce to empty translation units via their own
# #ifdefs, so the list is kept faithful rather than pruned. That makes diffing
# against a newer Vim's BASIC_SRC a mechanical check.
#
# Source is NOT in this repo: it is extracted and patched into build-deps/ by
# scripts/prepare-deps.sh. See docs/PLAN.md Phase 0.

cmake_minimum_required(VERSION 3.16)

set(VIM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../build-deps/vim")
set(VIM_SRC  "${VIM_ROOT}/src")

# vim.h includes "auto/config.h" with quotes, which searches vim.h's OWN
# directory before any -I path. Running Vim's configure on the host (pixi run
# configure-vim) writes a host config.h there, and the firmware would then be
# built with the HOST's settings -- silently. Refuse instead.
if(EXISTS "${VIM_SRC}/auto/config.h")
    message(FATAL_ERROR
        "${VIM_SRC}/auto/config.h exists: a HOST configure output would shadow "
        "port/auto/config.h and be compiled into the firmware.\n"
        "Run: pixi run deps   (re-extracts a clean tree)")
endif()

if(NOT EXISTS "${VIM_SRC}/main.c")
    message(FATAL_ERROR
        "Vim source not found at ${VIM_SRC}.\n"
        "Run: pixi run deps")
endif()

# XDIFF_SRC from Vim's Makefile. Vim bundles its own xdiff, which is what makes
# FEAT_DIFF work with no external `diff` binary -- essential here, since there
# is no shell to run one. Phase 8 reuses it for :EspGitDiff.
set(VIM_XDIFF_SOURCES
    "${VIM_SRC}/xdiff/xdiffi.c"
    "${VIM_SRC}/xdiff/xemit.c"
    "${VIM_SRC}/xdiff/xprepare.c"
    "${VIM_SRC}/xdiff/xutils.c"
    "${VIM_SRC}/xdiff/xhistogram.c"
    "${VIM_SRC}/xdiff/xpatience.c"
)

idf_component_register(
    SRCS "${VIM_SOURCES}" "${VIM_XDIFF_SOURCES}" "port/pathdef.c" "port/esp_shims.c"
         "port/esp_busy.c"
         "api/esp_api_sys.c" "api/esp_api_nvs.c" "api/esp_api_gpio.c"
    INCLUDE_DIRS "port"
    PRIV_INCLUDE_DIRS "${VIM_SRC}" "${VIM_SRC}/proto"   # upstream Makefile passes -Iproto
    REQUIRES fatfs vfs esp_partition esp_driver_uart esp_timer heap
             # the esp_*() builtins (api/)
             nvs_flash esp_driver_gpio esp_psram spi_flash esp_hw_support
    LDFRAGMENTS "linker.lf"     # brackets Vim's .data/.bss for session restarts
)

# Keep every Vim global inside the linker.lf brackets. RISC-V GCC otherwise
# puts small globals (<= 8 bytes: got_int, curbuf, ...) in .sdata/.sbss, which
# ESP-IDF's data/bss schemes do not cover -- a session restart would then leave
# exactly the most important state behind.
if(CONFIG_IDF_TARGET_ARCH_RISCV)
    target_compile_options(${COMPONENT_LIB} PRIVATE -msmall-data-limit=0)
endif()

# The chip's marketing name, for the intro screen ("Vim running on ESP32-P4",
# patches/vim/0007) and anything else that names the hardware: esp32p4 ->
# ESP32-P4, esp32s3 -> ESP32-S3. The runtime image's :help gets the same name
# from scripts/make-runtime-image.py.
if(IDF_TARGET MATCHES "^esp32(.+)$")
    string(TOUPPER "${CMAKE_MATCH_1}" chip_suffix)
    set(ESP_VIM_CHIP "ESP32-${chip_suffix}")
else()
    string(TOUPPER "${IDF_TARGET}" ESP_VIM_CHIP)
endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE ESP_VIM_CHIP="${ESP_VIM_CHIP}")

# -include forces our config.h ahead of everything, so Vim's sources need no
# edit to find it. HAVE_CONFIG_H makes vim.h look for auto/config.h, which we
# shadow via the port/ include dir taking precedence.
target_compile_options(${COMPONENT_LIB} PRIVATE
    -DHAVE_CONFIG_H
    -DESP_PLATFORM
    -Dmain=vim_main          # app_main owns the real entry point
    -include "${CMAKE_CURRENT_LIST_DIR}/port/auto/config.h"

    # Symbol collisions with ESP-IDF, found by diffing libvim.a's globals
    # against every IDF archive. Vim names its Vimscript builtins f_<name>,
    # which runs straight into FatFs's f_* API; timer_start collides with IDF's
    # own. Renaming via -D keeps both the definition and its callers consistent
    # inside this component, and needs no upstream patch.
    -Df_mkdir=vim_f_mkdir
    -Df_readdir=vim_f_readdir
    -Dtimer_start=vim_timer_start

    # Vim's heap goes to PSRAM. Left to IDF's default policy, every allocation
    # under CONFIG_SPIRAM_MALLOC_ALWAYSINTERNAL (16 KB) lands in internal RAM
    # first -- which is nearly all of Vim's allocations. Sourcing the real runtime
    # then exhausted internal RAM and ESP-IDF's own internal-only allocations
    # failed: flash reads returned ESP_ERR_NO_MEM and newlib aborted creating a
    # FILE lock. Renaming inside this component only keeps internal RAM for
    # FreeRTOS, drivers and (later) networking. free() needs no rename: it
    # releases memory from any heap.
    -Dmalloc=esp_vim_malloc
    -Dcalloc=esp_vim_calloc
    -Drealloc=esp_vim_realloc
    # free() too: Vim's blocks carry a tracking header (port/esp_shims.c) so a
    # new session can free them all; anything else goes to ESP-IDF's heap.
    -Dfree=esp_vim_free
)

# ESP-IDF defines chdir(), getcwd() and system(), so the port cannot simply
# define its own -- that is a duplicate symbol. --wrap lets us interpose:
# references resolve to __wrap_*, and __real_* reaches the original.
#
# This is also what makes rename/mkdir/opendir work. They have no mch_* macro to
# redirect (mch_rename is also a real function, so a macro mangles its
# prototype), and defining them under their own names would recurse infinitely.
# With --wrap the override calls __real_* instead.
set(VIM_WRAPPED
    # the userspace working directory: every path-taking call resolves
    chdir getcwd open fopen stat access unlink remove rmdir chmod rename mkdir opendir closedir
    # file identity: fstat must agree with stat (see esp_shims.c)
    fstat close fclose
    # process model, console, tty
    system exit _exit select tcgetattr
)
foreach(sym IN LISTS VIM_WRAPPED)
    target_link_libraries(${COMPONENT_LIB} INTERFACE "-Wl,--wrap=${sym}")
endforeach()

# Vim is a 30-year-old C codebase and builds cleanly under its own warning set,
# not IDF's. These are silenced deliberately rather than by patching upstream.
target_compile_options(${COMPONENT_LIB} PRIVATE
    -Wno-unused-parameter
    -Wno-unused-but-set-variable
    -Wno-sign-compare
    -Wno-implicit-fallthrough
    -Wno-missing-field-initializers
    -Wno-type-limits
    -Wno-char-subscripts
    -Wno-format-truncation
    -Wno-stringop-truncation
    -Wno-deprecated-non-prototype
    -Wno-old-style-declaration
    # IDF builds with -Werror=all; GCC's maybe-uninitialized is false-positive
    # prone at -Og and fires in upstream code we will not patch.
    -Wno-maybe-uninitialized
    -Wno-error=maybe-uninitialized
)
