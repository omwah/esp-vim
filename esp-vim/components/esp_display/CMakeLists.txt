# The console on an LCD (include/esp_display.h). Built from esp_display.c, the
# libvterm that ships inside Vim's source (build-deps/vim/src/libvterm), and
# font tables generated from the vendored Spleen and Terminus fonts
# (scripts/bdf2c.py).
set(VTERM "${CMAKE_CURRENT_LIST_DIR}/../../../build-deps/vim/src/libvterm")
set(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../build-deps")
set(BDF2C "${CMAKE_CURRENT_LIST_DIR}/../../../scripts/bdf2c.py")
# The overlay's font, as configured (Kconfig).
set(FONT_BIG_SIZE "12x24")
if(CONFIG_ESP_VIM_DISP_OVERLAY_16X32)
    set(FONT_BIG_SIZE "16x32")
endif()
set(FONT_BIG_BDF "${DEPS}/spleen/spleen-${FONT_BIG_SIZE}.bdf")
set(FONT_BIG_C "${CMAKE_CURRENT_BINARY_DIR}/esp_display_font_big.c")

if(CONFIG_ESP_VIM_DISPLAY)
    # The terminal's fonts (Kconfig ESP_VIM_DISP_FONTS): a table each, and a
    # list of them, the first the default.
    separate_arguments(FONTS UNIX_COMMAND "${CONFIG_ESP_VIM_DISP_FONTS}")
    if(NOT FONTS)
        message(FATAL_ERROR "ESP_VIM_DISP_FONTS lists no fonts")
    endif()
    set(FONT_SRCS "")
    set(FONT_DECLS "")
    set(FONT_ROWS "")
    set(i 0)
    foreach(font IN LISTS FONTS)
        if(font MATCHES "^spleen-([0-9]+x[0-9]+)$")
            set(bdf "${DEPS}/spleen/spleen-${CMAKE_MATCH_1}.bdf")
        elseif(font MATCHES "^terminus-[0-9]+x([0-9]+)$")
            set(bdf "${DEPS}/terminus/ter-u${CMAKE_MATCH_1}n.bdf")
        elseif(font MATCHES "^terminus-bold-[0-9]+x([0-9]+)$")
            set(bdf "${DEPS}/terminus/ter-u${CMAKE_MATCH_1}b.bdf")
        else()
            message(FATAL_ERROR "ESP_VIM_DISP_FONTS: unknown font '${font}' "
                "(spleen-WxH, terminus-WxH or terminus-bold-WxH)")
        endif()
        if(NOT EXISTS "${bdf}")
            message(FATAL_ERROR "Font ${font} not found at ${bdf}. Run: pixi run deps")
        endif()
        set(c "${CMAKE_CURRENT_BINARY_DIR}/esp_display_font_${i}.c")
        add_custom_command(
            OUTPUT "${c}"
            COMMAND ${python} "${BDF2C}" "${bdf}" "${c}" "esp_display_font_${i}" "${font}"
            DEPENDS "${bdf}" "${BDF2C}"
            VERBATIM)
        list(APPEND FONT_SRCS "${c}")
        string(APPEND FONT_DECLS "extern const esp_display_font_t esp_display_font_${i};\n")
        string(APPEND FONT_ROWS "    &esp_display_font_${i},\n")
        math(EXPR i "${i} + 1")
    endforeach()
    set(FONT_LIST_C "${CMAKE_CURRENT_BINARY_DIR}/esp_display_fonts.c")
    file(CONFIGURE OUTPUT "${FONT_LIST_C}" CONTENT
        "/* Generated by components/esp_display/CMakeLists.txt; do not edit. */\n#include \"esp_display_font.h\"\n\n${FONT_DECLS}\nconst esp_display_font_t *const esp_display_fonts[] = {\n${FONT_ROWS}};\nconst int esp_display_font_count = ${i};\n")
    if(NOT EXISTS "${FONT_BIG_BDF}")
        message(FATAL_ERROR "Spleen font not found at ${FONT_BIG_BDF}. Run: pixi run deps")
    endif()
    idf_component_register(
        SRCS "esp_display.c"
             "${VTERM}/src/encoding.c" "${VTERM}/src/keyboard.c" "${VTERM}/src/mouse.c"
             "${VTERM}/src/parser.c" "${VTERM}/src/pen.c" "${VTERM}/src/screen.c"
             "${VTERM}/src/state.c" "${VTERM}/src/unicode.c" "${VTERM}/src/vterm.c"
        INCLUDE_DIRS "include"
        PRIV_INCLUDE_DIRS "${VTERM}/include" "${VTERM}/src"
        PRIV_REQUIRES esp_lcd esp_driver_spi esp_driver_gpio heap log esp_kbd esp_touch esp_timer nvs_flash
    )
    # The overlay's (the Bluetooth pairing screen): big enough to tap.
    add_custom_command(
        OUTPUT "${FONT_BIG_C}"
        COMMAND ${python} "${BDF2C}" "${FONT_BIG_BDF}" "${FONT_BIG_C}" esp_display_font_big
        DEPENDS "${FONT_BIG_BDF}" "${BDF2C}"
        VERBATIM)
    target_sources(${COMPONENT_LIB} PRIVATE ${FONT_SRCS} "${FONT_LIST_C}" "${FONT_BIG_C}")
    # As Vim's Makefile builds it, but with libvterm's own character-width
    # tables rather than Vim's: the display task must not call into Vim.
    target_compile_definitions(${COMPONENT_LIB} PRIVATE INLINE=)
    # libvterm is upstream code, compiled as it is.
    set_source_files_properties(
        "${VTERM}/src/encoding.c" "${VTERM}/src/keyboard.c" "${VTERM}/src/mouse.c"
        "${VTERM}/src/parser.c" "${VTERM}/src/pen.c" "${VTERM}/src/screen.c"
        "${VTERM}/src/state.c" "${VTERM}/src/unicode.c" "${VTERM}/src/vterm.c"
        PROPERTIES COMPILE_OPTIONS "-Wno-unused-variable;-Wno-unused-function;-Wno-sign-compare;-Wno-unused-but-set-variable;-Wno-implicit-fallthrough;-Wno-enum-conversion;-Wno-type-limits;-Wno-missing-field-initializers;-Wno-char-subscripts")
else()
    # The same requirements as above: ESP-IDF works them out before the
    # configuration is known.
    idf_component_register(SRCS "esp_display_stub.c" INCLUDE_DIRS "include"
        PRIV_REQUIRES esp_lcd esp_driver_spi esp_driver_gpio heap log esp_kbd esp_touch esp_timer nvs_flash)
endif()
